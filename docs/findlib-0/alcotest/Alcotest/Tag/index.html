Module Alcotest.Tag

Combinators for tag, as in Tag.(slow && v "release").

val v : string -> tag

v s is `Tag s.

val quick : tag
val slow : tag
val always : tag
val (&&) : tag -> tag -> tag

a && b is `And [ a; b ].

val (||) : tag -> tag -> tag

a || b is `Or [ a; b ].

val not : tag -> tag

not a is `Not a.
